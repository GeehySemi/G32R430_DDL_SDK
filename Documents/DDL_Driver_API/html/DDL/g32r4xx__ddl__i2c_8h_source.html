780__STATIC_INLINE void DDL_I2C_ConfigSpeed(I2C_TypeDef *I2Cx, uint32_t PeriphClock, uint32_t ClockSpeed,
781 uint32_t DutyCycle)
782{
783 uint32_t freqrange = 0x0U;
784 uint32_t clockconfig = 0x0U;
785
786 /* Compute frequency range */
787 freqrange = __DDL_I2C_FREQ_HZ_TO_MHZ(PeriphClock);
788
789 /* Configure I2Cx: Frequency range register */
790 MODIFY_REG(I2Cx->CTRL2, I2C_CTRL2_CLKFCFG, freqrange);
791
792 /* Configure I2Cx: Rise Time register */
793 MODIFY_REG(I2Cx->RISETMAX, I2C_RISETMAX_RISETMAX, __DDL_I2C_RISE_TIME(freqrange, ClockSpeed));
794
795 /* Configure Speed mode, Duty Cycle and Clock control register value */
796 if (ClockSpeed > DDL_I2C_MAX_SPEED_STANDARD)
797 {
798 /* Set Speed mode at fast and duty cycle for Clock Speed request in fast clock range */
799 clockconfig = DDL_I2C_CLOCK_SPEED_FAST_MODE | \
800 __DDL_I2C_SPEED_FAST_TO_CCR(PeriphClock, ClockSpeed, DutyCycle) | \
801 DutyCycle;
802 }
803 else
804 {
805 /* Set Speed mode at standard for Clock Speed request in standard clock range */
806 clockconfig = DDL_I2C_CLOCK_SPEED_STANDARD_MODE | \
807 __DDL_I2C_SPEED_STANDARD_TO_CCR(PeriphClock, ClockSpeed);
808 }
809
810 /* Configure I2Cx: Clock control register */
811 MODIFY_REG(I2Cx->CLKCTRL, (I2C_CLKCTRL_SPEEDCFG | I2C_CLKCTRL_FDUTYCFG | I2C_CLKCTRL_CLKS), clockconfig);
812}

1225__STATIC_INLINE uint32_t DDL_I2C_ReadEventStatus(I2C_TypeDef *I2Cx, uint32_t i2cEvent)
1226{
1227 /* Read STS1 and STS2. STS1 lower 16 bits, STS2 lower 16 bits shifted left 16 */
1228 uint32_t flag1 = READ_BIT(I2Cx->STS1, 0x0000FFFFUL); /* Read all STS1 bits */
1229 uint32_t flag2 = READ_BIT(I2Cx->STS2, 0x0000FFFFUL); /* Read all STS2 bits */
1230
1231 /* Shift STS2 read result left by 16 to match the same position as i2cEvent */
1232 flag2 <<= 16;
1233
1234 /* Combine STS1 and STS2 as a 24-bit event status */
1235 uint32_t lastevent = (flag1 | flag2) & 0x00FFFFFFUL;
1236
1237 /* Check if all bits in i2cEvent are set */
1238 if ((lastevent & i2cEvent) == i2cEvent)
1239 {
1240 return 1UL;
1241 }
1242
1243 return 0UL;
1244}
