809{
810 uint32_t *reg = (uint32_t*)((uint32_t)(&(EINT->IOSELR1)) + ((!!(EintLine & ~(0xff))) << 2U));
811 uint32_t offset = ((POSITION_VAL(EintLine) & 0x07) << 2U);
