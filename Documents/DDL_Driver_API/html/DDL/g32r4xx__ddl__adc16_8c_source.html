80/* Definitions of ADC hardware constraints delays */
81/* Note: Only ADC peripheral HW delays are defined in ADC LL driver driver, */
82/* not timeout values: */
83/* Timeout values for ADC operations are dependent to device clock */
84/* configuration (system clock versus ADC clock), */
85/* and therefore must be defined in user application. */
86/* Refer to @ref ADC_LL_EC_HW_DELAYS for description of ADC timeout */
87/* values definition. */
88/* Note: ADC timeout values are defined here in CPU cycles to be independent */
89/* of device clock setting. */
90/* In user application, ADC timeout values should be defined with */
91/* temporal values, in function of device clock settings. */
92/* Highest ratio CPU clock frequency vs ADC clock frequency: */
93/* - ADC clock from synchronous clock with AHB prescaler 512, */
94/* ADC prescaler 4. */
95/* Ratio max = 512 *4 = 2048 */
96/* - ADC clock from asynchronous clock (PLLP) with prescaler 256. */
97/* Highest CPU clock PLL (PLLR). */
98/* Ratio max = PLLRmax /PPLPmin * 256 = (VCO/2) / (VCO/31) * 256 */
99/* = 3968 */
100/* Unit: CPU cycles. */
101#define ADC_CLOCK_RATIO_VS_CPU_HIGHEST (3968UL)
102#define ADC_TIMEOUT_DISABLE_CPU_CYCLES (ADC_CLOCK_RATIO_VS_CPU_HIGHEST * 1UL)
103#define ADC_TIMEOUT_STOP_CONVERSION_CPU_CYCLES (ADC_CLOCK_RATIO_VS_CPU_HIGHEST * 1UL)

250ErrorStatus DDL_ADC16_DeInit(ADC16_TypeDef *ADCx)
251{
252 ErrorStatus status = SUCCESS;
253
254 __IO uint32_t timeout_cpu_cycles = 0UL;
255
256 /* Check the parameters */
257 ASSERT_PARAM(IS_ADC16_ALL_INSTANCE(ADCx));
258
259 /* Disable ADC instance if not already disabled. */
260 if (DDL_ADC16_IsEnabled(ADCx) == 1UL)
261 {
262 /* Set ADC group regular trigger source to SW start to ensure to not */
263 /* have an external trigger event occurring during the conversion stop */
264 /* ADC disable process. */
265 DDL_ADC16_REG_SetTriggerSource(ADCx, DDL_ADC16_REG_TRIG_SOFTWARE);
266
267 /* Stop potential ADC conversion on going on ADC group regular. */
268 if (DDL_ADC16_REG_IsConversionOngoing(ADCx) != 0UL)
269 {
270 if (DDL_ADC16_REG_IsStopConversionOngoing(ADCx) == 0UL)
271 {
272 DDL_ADC16_REG_StopConversion(ADCx);
273 }
274 }
275
276 /* Set ADC group injected trigger source to SW start to ensure to not */
277 /* have an external trigger event occurring during the conversion stop */
278 /* ADC disable process. */
279 DDL_ADC16_INJ_SetTriggerSource(ADCx, DDL_ADC16_INJ_TRIG_SOFTWARE);
280
281 /* Stop potential ADC conversion on going on ADC group injected. */
282 if (DDL_ADC16_INJ_IsConversionOngoing(ADCx) != 0UL)
283 {
284 if (DDL_ADC16_INJ_IsStopConversionOngoing(ADCx) == 0UL)
285 {
286 DDL_ADC16_INJ_StopConversion(ADCx);
287 }
288 }
289
290 /* Wait for ADC conversions are effectively stopped */
291 timeout_cpu_cycles = ADC_TIMEOUT_STOP_CONVERSION_CPU_CYCLES;
292 while ((DDL_ADC16_REG_IsStopConversionOngoing(ADCx)
293 | DDL_ADC16_INJ_IsStopConversionOngoing(ADCx)) == 1UL)
294 {
295 timeout_cpu_cycles--;
296 if (timeout_cpu_cycles == 0UL)
297 {
298 /* Time-out error */
299 status = ERROR;
300 break;
301 }
302 }
303
304 /* Disable the ADC instance */
305 DDL_ADC16_Disable(ADCx);
306
307 }
308
309 /* Check whether ADC state is compliant with expected state */
310 if (READ_BIT(ADCx->CR,
311 (ADC_CR_JADSTP | ADC_CR_ADSTP | ADC_CR_JADSTART | ADC_CR_ADSTART
312 | ADC_CR_ADEN)
313 )
314 == 0UL)
315 {
316 /* ========== Reset ADC registers ========== */
317 /* Reset register IER */
318 CLEAR_BIT(ADCx->IER,
319 ( DDL_ADC16_IT_EOC
320 | DDL_ADC16_IT_EOS
321 | DDL_ADC16_IT_OVR
322 | DDL_ADC16_IT_EOSMP
323 | DDL_ADC16_IT_JEOC
324 | DDL_ADC16_IT_JEOS
325 | DDL_ADC16_IT_JQOVF
326 | DDL_ADC16_IT_AWD1
327 | DDL_ADC16_IT_AWD2
328 | DDL_ADC16_IT_AWD3
329 )
330 );
331
332 /* Reset register ISR */
333 SET_BIT(ADCx->ISR,
334 ( DDL_ADC16_FLAG_EOC
335 | DDL_ADC16_FLAG_EOS
336 | DDL_ADC16_FLAG_OVR
337 | DDL_ADC16_FLAG_EOSMP
338 | DDL_ADC16_FLAG_JEOC
339 | DDL_ADC16_FLAG_JEOS
340 | DDL_ADC16_FLAG_JQOVF
341 | DDL_ADC16_FLAG_AWD1
342 | DDL_ADC16_FLAG_AWD2
343 | DDL_ADC16_FLAG_AWD3
344 )
345 );
346
347 /* Reset register CR */
348 /* - Bits ADC_CR_JADSTP, ADC_CR_ADSTP, ADC_CR_JADSTART, ADC_CR_ADSTART, */
349 /* ADC_CR_ADCAL, ADC_CR_ADDIS, ADC_CR_ADEN are in */
350 /* access mode "read-set": no direct reset applicable. */
351 /* - Reset Calibration mode to default setting (single ended). */
352 /* - Disable ADC internal voltage regulator. */
353 /* - Enable ADC deep power down. */
354 /* Note: ADC internal voltage regulator disable and ADC deep power */
355 /* down enable are conditioned to ADC state disabled: */
356 /* already done above. */
357
358 /* Reset register CFGR */
359 MODIFY_REG(ADCx->CFGR,
360 ( ADC_CFGR_AWD1CH | ADC_CFGR_JAUTO | ADC_CFGR_JAWD1EN
361 | ADC_CFGR_AWD1EN | ADC_CFGR_AWD1SGL | ADC_CFGR_JQM
362 | ADC_CFGR_JDISCEN | ADC_CFGR_DISCNUM | ADC_CFGR_DISCEN
363 | ADC_CFGR_AUTDLY | ADC_CFGR_CONT | ADC_CFGR_OVRMOD
364 | ADC_CFGR_EXTEN | ADC_CFGR_EXTSEL
365 | ADC_CFGR_DMACFG | ADC_CFGR_DMAEN),
366 ADC_CFGR_JQDIS
367 );
368
369 /* Reset register CFGR2 */
370 CLEAR_BIT(ADCx->CFGR2,
371 ( ADC_CFGR2_DIFSEL | ADC_CFGR2_AWDFILT
372 | ADC_CFGR2_ROVSM | ADC_CFGR2_TROVS | ADC_CFGR2_OVSS
373 | ADC_CFGR2_OVSR | ADC_CFGR2_JOVSE | ADC_CFGR2_ROVSE)
374 );
375
376 /* Reset register SMPR1 */
377 CLEAR_BIT(ADCx->SMPR1,
378 ( ADC_SMPR1_SMP5 | ADC_SMPR1_SMP4 | ADC_SMPR1_SMP3
379 | ADC_SMPR1_SMP2 | ADC_SMPR1_SMP1 | ADC_SMPR1_SMP0)
380 );
381
382 /* Reset register TR1 */
383 MODIFY_REG(ADCx->TR1, ADC_TR1_HT1 | ADC_TR1_LT1, ADC_TR1_HT1);
384
385 /* Reset register TR2 */
386 MODIFY_REG(ADCx->TR2, ADC_TR2_HT2 | ADC_TR2_LT2, ADC_TR2_HT2);
387
388 /* Reset register TR3 */
389 MODIFY_REG(ADCx->TR3, ADC_TR3_HT3 | ADC_TR3_LT3, ADC_TR3_HT3);
390
391 /* Reset register SQR1 */
392 CLEAR_BIT(ADCx->SQR1,
393 ( ADC_SQR1_SQ7 | ADC_SQR1_SQ6 | ADC_SQR1_SQ5
394 | ADC_SQR1_SQ4 | ADC_SQR1_SQ3 | ADC_SQR1_SQ2
395 | ADC_SQR1_SQ1 | ADC_SQR1_RL)
396 );
397
398 /* Reset register SQR2 */
399 CLEAR_BIT(ADCx->SQR2,
400 ( ADC_SQR2_SQ12 | ADC_SQR2_SQ11 | ADC_SQR2_SQ10
401 | ADC_SQR2_SQ9 | ADC_SQR2_SQ8)
402 );
403
404 /* Reset register JSQR */
405 CLEAR_BIT(ADCx->JSQR,
406 (ADC_JSQR_JL
407 | ADC_JSQR_JEXTSEL | ADC_JSQR_JEXTEN
408 | ADC_JSQR_JSQ4 | ADC_JSQR_JSQ3
409 | ADC_JSQR_JSQ2 | ADC_JSQR_JSQ1)
410 );
411
412 /* Reset register DR */
413 /* Note: bits in access mode read only, no direct reset applicable */
414
415 /* Reset register OFFSET0 */
416 CLEAR_BIT(ADCx->OFFSET0, ADC_OFFSET0_OFFSET0_EN | ADC_OFFSET0_OFFSET0 | ADC_OFFSET0_OFFSETPOS);
417 /* Reset register OFFSET1 */
418 CLEAR_BIT(ADCx->OFFSET1, ADC_OFFSET1_OFFSET1_EN | ADC_OFFSET1_OFFSET1 | ADC_OFFSET1_OFFSETPOS);
419 /* Reset register OFFSET2 */
420 CLEAR_BIT(ADCx->OFFSET2, ADC_OFFSET2_OFFSET2_EN | ADC_OFFSET2_OFFSET2 | ADC_OFFSET2_OFFSETPOS);
421 /* Reset register OFFSET3 */
422 CLEAR_BIT(ADCx->OFFSET3, ADC_OFFSET3_OFFSET3_EN | ADC_OFFSET3_OFFSET3 | ADC_OFFSET3_OFFSETPOS);
423 /* Reset register OFFSET4 */
424 CLEAR_BIT(ADCx->OFFSET4, ADC_OFFSET4_OFFSET4_EN | ADC_OFFSET4_OFFSET4 | ADC_OFFSET4_OFFSETPOS);
425 /* Reset register OFFSET5 */
426 CLEAR_BIT(ADCx->OFFSET5, ADC_OFFSET5_OFFSET5_EN | ADC_OFFSET5_OFFSET5 | ADC_OFFSET5_OFFSETPOS);
427
428 /* Reset registers JDR1, JDR2, JDR3, JDR4 */
429 /* Note: bits in access mode read only, no direct reset applicable */
430
431 /* Reset register AWD2CR */
432 CLEAR_BIT(ADCx->AWD2CR, ADC_AWD2CR_AWD2CH);
433
434 /* Reset register AWD3CR */
435 CLEAR_BIT(ADCx->AWD3CR, ADC_AWD3CR_AWD3CH);
436
437 /* Reset register CAL */
438 CLEAR_BIT( ADCx->CAL,ADC_CAL_REF_SEL);
439 }
440 else
441 {
442 /* ADC instance is in an unknown state */
443 /* Need to performing a hard reset of ADC instance, using high level */
444 /* clock source RCM ADC reset. */
445 status = ERROR;
446 }
447
448 return status;
449}

557ErrorStatus DDL_ADC16_REG_Init(ADC16_TypeDef *ADCx, DDL_ADC16_REG_InitTypeDef *ADC_REG_InitStruct)
558{
559 ErrorStatus status = SUCCESS;
560
561 /* Check the parameters */
562 ASSERT_PARAM(IS_ADC16_ALL_INSTANCE(ADCx));
563 ASSERT_PARAM(IS_DDL_ADC16_REG_TRIG_SOURCE(ADCx, ADC_REG_InitStruct->TriggerSource));
564 ASSERT_PARAM(IS_DDL_ADC16_REG_SEQ_SCAN_LENGTH(ADC_REG_InitStruct->SequencerLength));
565 if (ADC_REG_InitStruct->SequencerLength != DDL_ADC16_REG_SEQ_SCAN_DISABLE)
566 {
567 ASSERT_PARAM(IS_DDL_ADC16_REG_SEQ_SCAN_DISCONT_MODE(ADC_REG_InitStruct->SequencerDiscont));
568
569 /* ADC group regular continuous mode and discontinuous mode */
570 /* can not be enabled simultenaeously */
571 ASSERT_PARAM((ADC_REG_InitStruct->ContinuousMode == DDL_ADC16_REG_CONV_SINGLE)
572 || (ADC_REG_InitStruct->SequencerDiscont == DDL_ADC16_REG_SEQ_DISCONT_DISABLE));
573 }
574 ASSERT_PARAM(IS_DDL_ADC16_REG_CONTINUOUS_MODE(ADC_REG_InitStruct->ContinuousMode));
575 ASSERT_PARAM(IS_DDL_ADC16_REG_DMA_TRANSFER(ADC_REG_InitStruct->DMATransfer));
576 ASSERT_PARAM(IS_DDL_ADC16_REG_OVR_DATA_BEHAVIOR(ADC_REG_InitStruct->Overrun));
577
578 /* Note: Hardware constraint (refer to description of this function): */
579 /* ADC instance must be disabled. */
580 if (DDL_ADC16_IsEnabled(ADCx) == 0UL)
581 {
582 /* Configuration of ADC hierarchical scope: */
583 /* - ADC group regular */
584 /* - Set ADC group regular trigger source */
585 /* - Set ADC group regular sequencer length */
586 /* - Set ADC group regular sequencer discontinuous mode */
587 /* - Set ADC group regular continuous mode */
588 /* - Set ADC group regular conversion data transfer: no transfer or */
589 /* transfer by DMA, and DMA requests mode */
590 /* - Set ADC group regular overrun behavior */
591 /* Note: On this G32R4 series, ADC trigger edge is set to value 0x0 by */
592 /* setting of trigger source to SW start. */
593 if (ADC_REG_InitStruct->SequencerLength != DDL_ADC16_REG_SEQ_SCAN_DISABLE)
594 {
595 MODIFY_REG(ADCx->CFGR,
596 ADC_CFGR_EXTSEL
597 | ADC_CFGR_EXTEN
598 | ADC_CFGR_DISCEN
599 | ADC_CFGR_DISCNUM
600 | ADC_CFGR_CONT
601 | ADC_CFGR_DMAEN
602 | ADC_CFGR_DMACFG
603 | ADC_CFGR_OVRMOD
604 ,
605 ADC_REG_InitStruct->TriggerSource
606 | ADC_REG_InitStruct->SequencerDiscont
607 | ADC_REG_InitStruct->ContinuousMode
608 | ADC_REG_InitStruct->DMATransfer
609 | ADC_REG_InitStruct->Overrun
610 );
611 }
612 else
613 {
614 MODIFY_REG(ADCx->CFGR,
615 ADC_CFGR_EXTSEL
616 | ADC_CFGR_EXTEN
617 | ADC_CFGR_DISCEN
618 | ADC_CFGR_DISCNUM
619 | ADC_CFGR_CONT
620 | ADC_CFGR_DMAEN
621 | ADC_CFGR_DMACFG
622 | ADC_CFGR_OVRMOD
623 ,
624 ADC_REG_InitStruct->TriggerSource
625 | DDL_ADC16_REG_SEQ_DISCONT_DISABLE
626 | ADC_REG_InitStruct->ContinuousMode
627 | ADC_REG_InitStruct->DMATransfer
628 | ADC_REG_InitStruct->Overrun
629 );
630 }
631
632 /* Set ADC group regular sequencer length and scan direction */
633 DDL_ADC16_REG_SetSequencerLength(ADCx, ADC_REG_InitStruct->SequencerLength);
634 }
635 else
636 {
637 /* Initialization error: ADC instance is not disabled. */
638 status = ERROR;
639 }
640 return status;
641}
