149ErrorStatus DDL_DAC_DeInit(DAC_TypeDef *DACx)
150{
151 /* Check the parameters */
152 ASSERT_PARAM(IS_DAC_ALL_INSTANCE(DACx));
153
154 DDL_RCM_Unlock();
155
156 if (DACx == DAC1)
157 {
158 /* Force reset of DAC clock */
159 DDL_RCM_EnableAPBReset(DDL_RCM_APB_RESET_DAC1);
160
161 /* Release reset of DAC clock */
162 DDL_RCM_DisableAPBReset(DDL_RCM_APB_RESET_DAC1);
163 }
164
165 if (DACx == DAC2)
166 {
167 /* Force reset of DAC clock */
168 DDL_RCM_EnableAPBReset(DDL_RCM_APB_RESET_DAC2);
169
170 /* Release reset of DAC clock */
171 DDL_RCM_DisableAPBReset(DDL_RCM_APB_RESET_DAC2);
172 }
173
174 return SUCCESS;
175}

196ErrorStatus DDL_DAC_Init(DAC_TypeDef *DACx, DDL_DAC_InitTypeDef *DAC_InitStruct)
197{
198 ErrorStatus status = SUCCESS;
199
200 /* Check the parameters */
201 ASSERT_PARAM(IS_DAC_ALL_INSTANCE(DACx));
202 ASSERT_PARAM(IS_DDL_DAC_TRIGGER_SOURCE(DAC_InitStruct->TriggerSource));
203 ASSERT_PARAM(IS_DDL_DAC_OUTPUT_BUFFER(DAC_InitStruct->OutputBuffer));
204 ASSERT_PARAM(IS_DDL_DAC_WAVE_AUTO_GENER_MODE(DAC_InitStruct->WaveAutoGeneration));
205 if (DAC_InitStruct->WaveAutoGeneration != DDL_DAC_WAVE_GENERATE_NONE)
206 {
207 ASSERT_PARAM(IS_DDL_DAC_WAVE_AUTO_GENER_CONFIG(DAC_InitStruct->WaveAutoGeneration,
208 DAC_InitStruct->WaveAutoGenerationConfig));
209 }
210
211 /* Note: Hardware constraint (refer to description of this function) */
212 /* DAC instance must be disabled. */
213 if (DDL_DAC_IsEnabled(DACx) == 0UL)
214 {
215 /* Configuration of DAC channel: */
216 /* - TriggerSource */
217 /* - WaveAutoGeneration */
218 /* - OutputBuffer */
219 /* - OutputMode */
220 if (DAC_InitStruct->WaveAutoGeneration != DDL_DAC_WAVE_GENERATE_NONE)
221 {
222 MODIFY_REG(DACx->CR,
223 (DAC_CR_TSEL
224 | DAC_CR_WAVE
225 | DAC_CR_MAMP
226 | DAC_CR_EXT_SEL
227 )
228 ,
229 (DAC_InitStruct->TriggerSource
230 | DAC_InitStruct->WaveAutoGeneration
231 | DAC_InitStruct->WaveAutoGenerationConfig
232 | DAC_InitStruct->OutputBuffer
233 )
234 );
235 }
236 else
237 {
238 MODIFY_REG(DACx->CR,
239 (DAC_CR_TSEL
240 | DAC_CR_WAVE
241 | DAC_CR_EXT_SEL
242 )
243 ,
244 (DAC_InitStruct->TriggerSource
245 | DDL_DAC_WAVE_GENERATE_NONE
246 | DAC_InitStruct->OutputBuffer
247 )
248 );
249 }
250 }
251 else
252 {
253 /* Initialization error: DAC instance is not disabled. */
254 status = ERROR;
255 }
256 return status;
257}

265void DDL_DAC_StructInit(DDL_DAC_InitTypeDef *DAC_InitStruct)
266{
267 /* Set DAC_InitStruct fields to default values */
268 DAC_InitStruct->TriggerSource = DDL_DAC_CHANNEL_TRIG_SOURCE_SOFTWARE;
269 DAC_InitStruct->WaveAutoGeneration = DDL_DAC_WAVE_GENERATE_NONE;
