320{

330__STATIC_INLINE uint32_t DDL_RTC_GetTime(RTC_TypeDef* RTCx)
331{
332 uint16_t cnt_high;
333 uint16_t cnt_low;
334
335 cnt_high = READ_BIT(RTCx->CNTH, RTC_CNT_CNTH);
336 cnt_low = READ_BIT(RTCx->CNTL, RTC_CNT_CNTL);
337
338 return ((uint32_t)cnt_high << 16U) | (uint32_t)cnt_low;
339}
