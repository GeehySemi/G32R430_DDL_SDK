152ErrorStatus DDL_FLASH_EraseMain(uint32_t type, uint32_t baseAddr)
153{
154 volatile uint32_t timeout = DDL_FLASH_TIMEOUT_VALUE;
155 ErrorStatus status = SUCCESS;
156
157 /* Check the parameters */
158 ASSERT_PARAM(IS_DDL_FLASH_ERASE_TYPE(type));
159 ASSERT_PARAM(IS_DDL_FLASH_MAIN_BASE_ADDRESS(baseAddr));
160
161 /* Check the FLASH_SR->BUSY bit to confirm that the last programming operation has ended. */
162 if (DDL_FLASH_IsActiveFlag_BUSY() != RESET)
163 {
164 status = ERROR;
165 return status;
166 }
167
168 /* Check FLASH_SR->MAINUNLOCK is 1. */
169 if (DDL_FLASH_IsActiveFlag_MAINUNLOCK() == RESET)
170 {
171 status = ERROR;
172 return status;
173 }
174
175 /* Set the erasure type of flash. */
176 DDL_FLASH_SetErasureType(type);
177
178 /* Set the EREQ bit of FLASH_CR to 1. */
179 DDL_FLASH_EnableErasure();
180
181 /* Write 0xA5A5 to baseAddr. */
182 __DDL_FLASH_WRITE_TO_ADDRESS(baseAddr, 0xA5A5);
183
184 /* Wait for the FLASH_SR->BSY bit to be cleared. */
185 timeout = DDL_FLASH_TIMEOUT_VALUE;
186 while(DDL_FLASH_IsActiveFlag_BUSY() != RESET)
187 {
188 if (timeout-- == 0)
189 {
190 status = ERROR;
191 return status;
192 }
193 }
194
195 /* Wait for the erasing to be completed. */
196 timeout = DDL_FLASH_TIMEOUT_VALUE;
197 while(DDL_FLASH_IsActiveFlag_ERD() == RESET)
198 {
199 if (timeout-- == 0)
200 {
201 status = ERROR;
202 return status;
203 }
204 }
205
206 /* Clear the erase completion flag. */
207 DDL_FLASH_ClearFlag_ERD();
208
209 if (DDL_FLASH_IsActiveFlag_WADRERR() != RESET)
210 {
211 status = ERROR;
212 }
213
214 if (DDL_FLASH_IsActiveFlag_WRPRTERR() != RESET)
215 {
216 status = ERROR;
217 }
218
219 return status;
220}

278ErrorStatus DDL_FLASH_OptionByteProgram(DDL_FLASH_OPT_InitTypeDef* FLASH_OPT_InitStruct)
279{
280 volatile uint32_t timeout = DDL_FLASH_TIMEOUT_VALUE;
281 ErrorStatus status = SUCCESS;
282
283 /* Check the parameters */
284 ASSERT_PARAM(IS_DDL_FLASH_READ_PROTECTION_LEVEL(FLASH_OPT_InitStruct->readProtectionLevel));
285 ASSERT_PARAM(IS_DDL_FLASH_IWDT_SOFTWARE_ENABLE(FLASH_OPT_InitStruct->iwdtSoftwareEnable));
286 ASSERT_PARAM(IS_DDL_FLASH_WWDT_SOFTWARE_ENABLE(FLASH_OPT_InitStruct->wwdtSoftwareEnable));
287 ASSERT_PARAM(IS_DDL_FLASH_WRITE_PROTECTION(FLASH_OPT_InitStruct->writeProtectionEnable));
288 ASSERT_PARAM(IS_DDL_FLASH_NMI_ENABLE(FLASH_OPT_InitStruct->nmiEnable));
289 ASSERT_PARAM(IS_DDL_FLASH_ADC_TRIM(FLASH_OPT_InitStruct->adcTrim));
290 ASSERT_PARAM(IS_DDL_FLASH_WRITE_PROTECTION_SECTORS(FLASH_OPT_InitStruct->writeProtectionSectors));
291
292 /* Read RDP from NVR1 region at 0x1FFF0000. */
293 (void)*((volatile uint32_t *)0x1FFF0000U);
294 __DMB();
295
296 /* Check the FLASH_SR->BUSY bit to confirm that the last programming operation has ended. */
297 if (DDL_FLASH_IsActiveFlag_BUSY() != RESET)
298 {
299 status = ERROR;
300 return status;
301 }
302
303 /* Check FLASH_SR->OPUNLOCK is 1. */
304 if (DDL_FLASH_IsActiveFlag_OPUNLOCK() == RESET)
305 {
306 status = ERROR;
307 return status;
308 }
309
310 /* Fill data to register OBCR1. */
311 MODIFY_REG(FLASH->OBCR1,
312 FLASH_OBCR1_RDPWBF \
313 | FLASH_OBCR1_IWDTSWBF \
314 | FLASH_OBCR1_WWDTSWBF \
315 | FLASH_OBCR1_WLOCKENBF \
316 | FLASH_OBCR1_NMIDISWBF \
317 | FLASH_OBCR1_ADTSLOADWBF \
318 | FLASH_OBCR1_WLOCKBF ,
319 FLASH_OPT_InitStruct->readProtectionLevel \
320 | FLASH_OPT_InitStruct->iwdtSoftwareEnable \
321 | FLASH_OPT_InitStruct->wwdtSoftwareEnable \
322 | FLASH_OPT_InitStruct->writeProtectionEnable \
323 | FLASH_OPT_InitStruct->adcTrim \
324 | FLASH_OPT_InitStruct->nmiEnable \
325 | (~FLASH_OPT_InitStruct->writeProtectionSectors) & 0xFF0000);
326
327 /* Checking the validity of the data. */
328 uint32_t actualAddr = DDL_FLASH_GetBootAddrTransferActualAddress();
329
330 if (__DDL_FLASH_IS_VALID_BOOT_ADDRESS(actualAddr) == RESET)
331 {
332 status = ERROR;
333 return status;
334 }
335
336 /* If the data is valid, fill the data into the register. */
337 MODIFY_REG(FLASH->OBCR2, FLASH_OBCR2_BOOTADDRBF, FLASH_OPT_InitStruct->bootAddr);
338
339 DDL_FLASH_OptionByteUpdate();
340
341 /* Wait for the FLASH_SR->BSY bit to be cleared. */
342 timeout = DDL_FLASH_TIMEOUT_VALUE;
343 while(DDL_FLASH_IsActiveFlag_BUSY() != RESET)
344 {
345 if (timeout-- == 0)
346 {
347 status = ERROR;
348 return status;
349 }
350 }
351
352 /* Wait for the programing to be completed. */
353 timeout = DDL_FLASH_TIMEOUT_VALUE;
354 while(DDL_FLASH_IsActiveFlag_OPBPRD() == RESET)
355 {
356 if (timeout-- == 0)
357 {
358 status = ERROR;
359 return status;
360 }
361 }
362
363 /* Re-power on or write 1 to opreload, FLS_OBSR1/1 status updates. */
364 DDL_FLASH_OptionByteForcedUpdate();
365
366 return status;
367}

380ErrorStatus DDL_FLASH_MainProgram(uint32_t baseAddr, uint32_t *data, uint32_t numWords)
381{
382 volatile uint32_t timeout = DDL_FLASH_TIMEOUT_VALUE;
383 ErrorStatus status = SUCCESS;
384
385 /* Check the FLASH_SR->BUSY bit to confirm that the last programming operation has ended. */
386 if (DDL_FLASH_IsActiveFlag_BUSY() != RESET)
387 {
388 status = ERROR;
389 return status;
390 }
391
392 /* Write data to Flash */
393 for (uint32_t i = 0; i < numWords; i++)
394 {
395 /* Check FLASH_SR->MAINUNLOCK is 1. */
396 if (DDL_FLASH_IsActiveFlag_MAINUNLOCK() == RESET)
397 {
398 status = ERROR;
399 return status;
400 }
401
402 /* Set the PREQ bit of FLASH_CR to 1. */
403 DDL_FLASH_EnableProgram();
404
405 *((uint32_t*)(baseAddr + i * sizeof(uint32_t))) = data[i];
406 /* Wait for the FLASH_SR->BSY bit to be cleared. */
407 timeout = DDL_FLASH_TIMEOUT_VALUE;
408 while(DDL_FLASH_IsActiveFlag_BUSY() != RESET)
409 {
410 if (timeout-- == 0)
411 {
412 status = ERROR;
413 return status;
414 }
415 }
416
417 /* Wait for the programing to be completed. */
418 timeout = DDL_FLASH_TIMEOUT_VALUE;
419 while(DDL_FLASH_IsActiveFlag_PRD() == RESET)
420 {
421 if (timeout-- == 0)
422 {
423 status = ERROR;
424 return status;
425 }
426 }
427
428 DDL_FLASH_ClearFlag_PRD();
429 }
430
431
432 if (DDL_FLASH_IsActiveFlag_WADRERR() != RESET)
433 {
434 status = ERROR;
435 return status;
436 }
437
438 if (DDL_FLASH_IsActiveFlag_WRPRTERR() != RESET)
439 {
440 status = ERROR;
441 return status;
442 }
443 return status;
444}
