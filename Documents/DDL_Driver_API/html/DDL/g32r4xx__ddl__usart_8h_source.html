1056__STATIC_INLINE void DDL_USART_SetBaudRate(USART_TypeDef *USARTx, uint32_t PeriphClk, uint32_t OverSampling,
1057 uint32_t BaudRate)
1058{
1059 uint32_t temp, intDiv, fractionalDiv;
1060 if (OverSampling == DDL_USART_OVERSAMPLING_8)
1061 {
1062 intDiv = ((25 * PeriphClk) / (2 * BaudRate));
1063 }
1064 else
1065 {
1066 intDiv = ((25 * PeriphClk) / (4 * BaudRate));
1067 }
1068
1069 temp = (intDiv / 100) << 4;
1070 fractionalDiv = intDiv - (100 * (temp >> 4));
1071
1072 if (OverSampling == DDL_USART_OVERSAMPLING_8)
1073 {
1074 temp |= ((((fractionalDiv * 8) + 50) / 100)) & ((uint8_t)0x07);
1075 }
1076 else
1077 {
1078 temp |= ((((fractionalDiv * 16) + 50) / 100)) & ((uint8_t)0x0F);
1079 }
1080
1081 USARTx->BR = temp;
1082}
